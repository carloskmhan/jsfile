/** Core release checks. Old expectations are NOT silently rewritten to force a green result. */
import {spawnSync} from 'node:child_process';import fs from 'node:fs';import path from 'node:path';import {ROOT} from '../tools/evaluate.mjs';
const python=process.env.PYTHON||'python',runs=[];
function run(cmd,args,allowDifference=false){const r=spawnSync(cmd,args,{cwd:ROOT,encoding:'utf8'});const name=args.join(' ');fs.writeFileSync(path.join(ROOT,'reports',name.replace(/[^a-z0-9]/gi,'_')+'.log'),(r.stdout||'')+(r.stderr||''));runs.push({name,status:r.status,expectedLegacyMigrationDifference:allowDifference});if(r.error||r.status!==0&&!allowDifference){console.error(name,r.error||r.stderr||r.stdout);process.exitCode=1;}else console.log(name,allowDifference?'(legacy expectations inspected below)':'PASS');}
run(python,['build_command_patterns.py','--check']);run(python,['tests/test_compiler.py']);run(python,['tests/test_holdout_guard.py']);run('node',['tests/test_engine.mjs']);run('node',['tools/evaluate.mjs']);run('node',['tests/test_robustness.mjs']);run('node',['tools/evaluate.mjs','tests/robustness.csv','reports/robustness.json']);run('node',['tools/fuzzy_clean_check.mjs']);
for(const name of ['legacy_regression.json','dialogue_application_regression.json'])fs.rmSync(path.join(ROOT,'reports',name),{force:true});
run('node',['tests/legacy_regression.mjs'],true);run('node',['tests/legacy_dialogue.mjs'],true);
const old=JSON.parse(fs.readFileSync(path.join(ROOT,'reports/legacy_regression.json'),'utf8')),dialogue=JSON.parse(fs.readFileSync(path.join(ROOT,'reports/dialogue_application_regression.json'),'utf8'));
const failures=old.outcomes.filter(x=>x.status!=='pass'),df=dialogue.results.filter(x=>!x.ok);
const expected=['D01:8','D01:9','D07:3','D07:4','D08:3'];
const known=failures.length===1&&failures[0].name==='Negated rhetorical question'&&failures[0].error.includes('Checking the positive proposition')&&df.length===5&&df.every(x=>expected.includes(x.id+':'+x.turn)&&(/Two separate periods require a comparison report|There is no matching reference/.test(x.answer)));
if(!known){console.error('Unexpected legacy regression; inspect raw reports.');process.exitCode=1;}
const review={suite:'v3-to-v5 expectation migration, not erased failures',legacyAccounting:{passed:old.passed,oldExpectationDifferences:old.failed},legacyDialogues:{passed:dialogue.passed,oldExpectationDifferences:dialogue.failed},allDifferencesExplicitlyExplained:known,changes:[{test:'Negated rhetorical question',policy:'Qualify the positive proposition before Yes/No; old /^Yes/ text assertion intentionally differs.'},{tests:['D01:8','D01:9','D07:3','D07:4'],policy:'After a two-period comparison, ask which month instead of silently falling back to screen month.'},{test:'D08:3',policy:'After the final ranked entity, reject next instead of repeating the last entity.'}],runs};
fs.writeFileSync(path.join(ROOT,'reports/migration_tests.json'),JSON.stringify(review,null,2));console.log(known?'All six old expectation differences are the explicitly tested safer behavior changes.':'Migration check FAILED');
