import fs from 'node:fs';import assert from 'node:assert/strict';
import {RwaQaEngine} from '../rwa_engine.js';import {RuleClient} from '../rule_client.js';
import {parseRuleText} from '../semantic/registry.js';import {planDataRequest} from '../semantic/routing.js';
import {parseCsv} from '../csv_adapter.js';import {tableauRowToClient,readWorksheetGroup,readWorksheetRows} from '../tableau_adapter.js';
import {loadFixtures} from '../tools/evaluate.mjs';
const root=new URL('../',import.meta.url),read=n=>fs.readFileSync(new URL(n,root),'utf8');
const registry=parseRuleText(read('command_patterns.txt')),catalog=JSON.parse(read('rwa_sample_data.txt')),config=JSON.parse(read('tableau_config.txt')),data=loadFixtures();
const options={...data,commandPatterns:registry,semanticCatalog:catalog,portfolioComplete:true};
const make=()=>new RwaQaEngine(data.rows,options),ctx={selectedClientGroup:'SAMSUNG GROUP',selectedMonth:'2026-06'};
const source=parseCsv(read('tableau_sample.csv')),clients=source.map(r=>tableauRowToClient(r,catalog.groups.find(g=>g.client_group_id===r.client_group_id),config.fields));
const results=[];
function test(name,fn){try{fn();results.push({test:name,passed:true});}catch(e){results.push({test:name,passed:false,error:e.stack});}}
async function asyncTest(name,fn){try{await fn();results.push({test:name,passed:true});}catch(e){results.push({test:name,passed:false,error:e.stack});}}
function ok(engine,q,c=ctx){const r=engine.answer(q,c);assert.equal(r.ok,true,q+': '+r.answer);return r;}
function client(){const c=new RuleClient({commandPatterns:registry,semanticCatalog:catalog});c.setData([clients[0]]);return c;}

test('Exact Samsung → Toyota → August patches and new data',()=>{const e=make();ok(e,'Why did Samsung’s RWA increase in July?');const r=ok(e,'How about Toyota?');assert.equal(r.plan.groupId,'CG0004');assert.equal(r.result.change,-30);const a=ok(e,'And August?');assert.equal(a.plan.groupId,'CG0004');assert.equal(a.period.month,'2026-08');assert.equal(a.result.change,70);});
test('July decrease reported despite increase wording',()=>{const r=ok(make(),'Why did Samsung RWA increase in July?');assert.equal(r.result.change,-120);assert.match(r.answer,/decreased/);});
test('State has only identity references, never financial values',()=>{const e=make();ok(e,'Top 10 groups');assert.ok(e.state.references.length);const s=JSON.stringify(e.state);for(const key of ['rwa_prev','rwa_curr','result','impact','value','changeDifference','shareOfAligned'])assert.equal(s.includes('"'+key+'"'),false,key);});

test('Runtime alias collision is ambiguous rather than first-match',()=>{const reg=JSON.parse(JSON.stringify(registry));reg.aliases.push({kind:'GROUP',canonical_id:'CG0004',canonical_name:'TOYOTA GROUP',alias:'samsung',enabled:true});const e=new RwaQaEngine(data.rows,{...options,commandPatterns:reg});assert.equal(e.parseQuestion('Explain Samsung',ctx).code,'AMBIGUOUS_ENTITY');});
test('Entity exact boundary: SEC does not match second',()=>{const reg=JSON.parse(JSON.stringify(registry));reg.aliases.push({kind:'ENTITY',canonical_id:'CG0001-E001',parent_id:'CG0001',alias:'sec',enabled:true});const e=new RwaQaEngine(data.rows,{...options,commandPatterns:reg});ok(e,'Top 3 entities');const r=ok(e,'What about the second one?');assert.equal(r.entity,'Samsung SDI');});
test('Same-score candidates are not arbitrarily selected',()=>{const reg=JSON.parse(JSON.stringify(registry));reg.commands.push({...reg.commands[0],command_id:'COMPETING_ROOT'});const e=new RwaQaEngine(data.rows,{...options,commandPatterns:reg});assert.equal(e.parseQuestion('Explain Samsung',ctx).code,'AMBIGUOUS_COMMAND');});
test('Weakly weighted known synonym below minimum cannot execute',()=>{const reg=JSON.parse(JSON.stringify(registry));for(const s of reg.synonyms)if(s.phrase==='explain')s.weight=.1;const e=new RwaQaEngine(data.rows,{...options,commandPatterns:reg});assert.equal(e.parseQuestion('Explain Samsung',ctx).code,'LOW_CONFIDENCE');});
test('Conflicting intentions receive penalty and rejection',()=>{const r=make().parseQuestion('Why Samsung RWA and compare Toyota?',ctx);assert.equal(r.ok,false);assert.ok(r.explain.candidates.some(c=>c.penalty<0));});
test('Known metric does not bypass forecast rejection',()=>{assert.equal(make().parseQuestion('Predict Samsung RWA next year.',ctx).code,'UNSUPPORTED_INTENT');});
test('Unknown modifiers not silently discarded',()=>{const e=make();const old=JSON.stringify(e.state);const r=e.answer('Top entities for Samsung with secret weighting',ctx);assert.equal(r.ok,false);assert.equal(JSON.stringify(e.state),old);});
test('Subject excluded from portfolio not selected as the requested group',()=>{const r=ok(make(),'Top 10 groups excluding Samsung');assert.equal(r.plan.group,null);assert.ok(r.plan.excludedGroupIds.includes('CG0001'));assert.ok(r.result.items.every(x=>x.name!=='SAMSUNG GROUP'));});

test('Negated month rejected',()=>{assert.equal(make().parseQuestion('Explain Samsung not in July',ctx).ok,false);});
test('Not only is not interpreted as sole-driver check',()=>{assert.equal(make().parseQuestion('Was it not only CG for Samsung?',ctx).ok,false);});
test('Rhetorical negative check is labelled as a positive proposition',()=>{const r=ok(make(),"Wasn't CG the main driver for Samsung?");assert.match(r.answer,/positive proposition about CG/);});
test('Compare first ranked group with Toyota resolves IDs and reload plan',()=>{const e=make();ok(e,'Top 10 groups');ok(e,'Only those above 25m');const req=planDataRequest('Compare the first one with Toyota',catalog,registry,e.state);assert.deepEqual(req.groupIds.sort(),['CG0001','CG0004']);assert.equal(req.portfolio,false);const r=ok(e,'Compare the first one with Toyota');assert.deepEqual(r.plan.comparisonIds,['CG0001','CG0004']);});

test('Explicit period supersedes screen and prior period',()=>{const e=make();ok(e,'Explain Samsung in June');const r=ok(e,'How about Toyota in July?');assert.equal(r.period.month,'2026-07');});
test('Complete new question resets previous exclusions',()=>{const e=make();ok(e,'Explain Samsung');ok(e,'Exclude FX');assert.deepEqual(ok(e,'Explain Toyota in June').plan.excludedDrivers,[]);});

test('Numeric filter kept only as user parameter; subset IDs have no values',()=>{const e=make();ok(e,'Top 3 entities');const r=ok(e,'Only those above 200m');assert.equal(r.result.items.length,2);assert.equal(e.state.references.length,2);assert.equal(e.state.references.some(x=>'value' in x),false);});

function fakeTable(rows,limited=false){const names=Object.values(config.fields);return{getColumns:()=>names.map((n,i)=>({getFieldName:()=>n,getIndex:()=>i})),getData:()=>rows.map(r=>names.map(n=>({value:r[n]}))),getTotalRowCount:()=>rows.length,getIsTotalRowCountLimited:()=>limited};}
await asyncTest('Original Tableau API uses ID filter and preserves JSON contract',async()=>{let args;const w={applyFilterAsync:async(...a)=>args=a,getSummaryDataAsync:async()=>fakeTable([source[0]])};const r=await readWorksheetGroup(w,catalog.groups[0],config);assert.equal(r.client_group_id,'CG0001');assert.equal(args[0],'client_group_id');assert.deepEqual(args[1],['CG0001']);});
